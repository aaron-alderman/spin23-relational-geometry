test/semantic-mutation-execution.test.mjs
import assert from "node:assert/strict";
import test from "node:test";
import {
  effectivePropertyMutationOperation,
  hasPresentMutationValue,
  isPropertyMutationOperation,
  mutationDeltaFact,
} from "../sdk/core/semantic-mutation-execution.mjs";

test("property mutation family is authored without admitting context operations", () => {
  for (const operation of ["setProperty", "appendProperty", "clearProperty", "setFact", "tombstoneFact", "addFact", "removeFact", "detachRelation"]) {
    assert.equal(isPropertyMutationOperation(operation), true);
  }
  assert.equal(isPropertyMutationOperation("attachContext"), false);
  assert.equal(isPropertyMutationOperation("unknown"), false);
});
test("property mutation facts author polarity, metadata absence, and exact value presence", () => {
  assert.deepEqual(mutationDeltaFact("s", "p", "o"), { s: "s", p: "p", o: "o", pol: "is" });
  assert.deepEqual(mutationDeltaFact("s", "p", "o", "is-not", { source: "test" }), {
    s: "s", p: "p", o: "o", pol: "is-not", source: "test",
  });
  assert.equal(hasPresentMutationValue({ value: undefined }), true);
  assert.equal(hasPresentMutationValue({}), false);
  assert.equal(hasPresentMutationValue(null), false);
});

test("effective property operation preserves explicit cardinality symmetry", () => {
  assert.equal(effectivePropertyMutationOperation("setFact", "many"), "appendProperty");
  assert.equal(effectivePropertyMutationOperation("setFact", "one"), "setProperty");
  assert.equal(effectivePropertyMutationOperation("addFact", "min1"), "appendProperty");
  assert.equal(effectivePropertyMutationOperation("addFact", "optional"), "setProperty");
  assert.equal(effectivePropertyMutationOperation("tombstoneFact", "one"), "clearProperty");
  assert.equal(effectivePropertyMutationOperation("removeFact", "many"), "removeFact");
});
